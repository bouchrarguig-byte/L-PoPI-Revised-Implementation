     1	import pandas as pd
     2	import torch
     3	import torch.nn.functional as F
     4	import os
     5	import glob
     6	import numpy as np
     7	from torch_geometric.data import Data
     8	from torch_geometric.nn import GATv2Conv # Using GATv2 for more dynamic attention
     9	from sklearn.metrics import classification_report
    10	from sklearn.preprocessing import StandardScaler
    11	
    12	# ==========================================
    13	# 1. THE ADVANCED COGNITIVE GAT (GATv2)
    14	# ==========================================
    15	class CognitiveGAT(torch.nn.Module):
    16	    def __init__(self, in_channels, out_channels):
    17	        super(CognitiveGAT, self).__init__()
    18	        # GATv2 is more robust to noise and better for imbalanced graphs
    19	        self.conv1 = GATv2Conv(in_channels, 32, heads=8, concat=True)
    20	        self.conv2 = GATv2Conv(32 * 8, 16, heads=8, concat=True)
    21	        self.conv3 = GATv2Conv(16 * 8, out_channels, heads=1, concat=False)
    22	        
    23	        # Residual projection to handle feature depth
    24	        self.residual = torch.nn.Linear(in_channels, 16 * 8)
    25	
    26	    def forward(self, data):
    27	        x, edge_index = data.x, data.edge_index
    28	        
    29	        # Layer 1
    30	        x1 = F.elu(self.conv1(x, edge_index))
    31	        
    32	        # Layer 2 + Residual Connection (Crucial for high accuracy)
    33	        x2 = F.elu(self.conv2(x1, edge_index))
    34	        res = self.residual(x)
    35	        x2 = x2 + res # Skip connection prevents information loss
    36	        
    37	        # Layer 3 (Decision Layer)
    38	        logits = self.conv3(x2, edge_index)
    39	        return F.log_softmax(logits, dim=1)
    40	
    41	# ==========================================
    42	# 2. FEATURE-STABILIZED LOADER
    43	# ==========================================
    44	class BoTIoTFullLoader:
    45	    def __init__(self, base_path):
    46	        self.base_path = base_path
    47	        self.ip_map = {}
    48	        self.node_count = 0
    49	        self.scaler = StandardScaler()
    50	        self.is_fitted = False
    51	
    52	    def get_node_id(self, ip):
    53	        ip_str = str(ip)
    54	        if ip_str not in self.ip_map:
    55	            self.ip_map[ip_str] = self.node_count
    56	            self.node_count += 1
    57	        return self.ip_map[ip_str]
    58	
    59	    def stream_snapshots(self):
    60	        all_csvs = sorted(glob.glob(os.path.join(self.base_path, "*.csv")))
    61	        data_files = [f for f in all_csvs if "data_names" not in f]
    62	
    63	        for file in data_files:
    64	            # Increased sample size to provide meaningful support
    65	            df = pd.read_csv(file, low_memory=False).sample(n=50000, random_state=42, replace=True)
    66	            df['attack'] = pd.to_numeric(df['attack'], errors='coerce').fillna(0).astype(int)
    67	            
    68	            # Strict 1:1 Balancing
    69	            benign = df[df['attack'] == 0]
    70	            attack = df[df['attack'] == 1]
    71	            min_len = min(len(benign), len(attack))
    72	            if min_len < 100: continue 
    73	            
    74	            df_balanced = pd.concat([benign.sample(min_len), attack.sample(min_len)]).reset_index()
    75	            
    76	            # Node Mapping
    77	            src_ids = [self.get_node_id(ip) for ip in df_balanced['saddr']]
    78	            dst_ids = [self.get_node_id(ip) for ip in df_balanced['daddr']]
    79	            edge_index = torch.tensor([src_ids, dst_ids], dtype=torch.long)
    80	
    81	            # Feature Scaling (Crucial for Publication Accuracy)
    82	            cols = ['pkts', 'bytes', 'seq', 'dur']
    83	            feats = df_balanced[cols].apply(pd.to_numeric).fillna(0).values
    84	            if not self.is_fitted:
    85	                self.scaler.fit(feats)
    86	                self.is_fitted = True
    87	            feats_scaled = self.scaler.transform(feats)
    88	
    89	            num_nodes = self.node_count
    90	            x_full = torch.zeros((num_nodes, 4))
    91	            active_ids = np.array(src_ids)
    92	            x_full[active_ids] = torch.from_numpy(feats_scaled).float()
    93	            
    94	            # Labeling
    95	            y = torch.zeros(num_nodes, dtype=torch.long)
    96	            y[active_ids] = torch.tensor(df_balanced['attack'].values, dtype=torch.long)
    97	            
    98	            yield Data(x=x_full, edge_index=edge_index, y=y), active_ids
    99	
   100	# ==========================================
   101	# 3. HIGH-PRECISION TRAINING
   102	# ==========================================
   103	if __name__ == "__main__":
   104	    DATA_PATH = "/home/rguig/BoT-IoT"
   105	    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
   106	    
   107	    loader = BoTIoTFullLoader(DATA_PATH)
   108	    model = CognitiveGAT(in_channels=4, out_channels=2).to(device)
   109	    optimizer = torch.optim.Adam(model.parameters(), lr=0.001, weight_decay=1e-5)
   110	    
   111	    all_preds, all_true = [], []
   112	
   113	    print(f"Executing High-Accuracy Cog-GAT on {device}...")
   114	
   115	    for i, (snapshot, active_ids) in enumerate(loader.stream_snapshots()):
   116	        snapshot = snapshot.to(device)
   117	        model.train()
   118	        optimizer.zero_grad()
   119	        
   120	        logits = model(snapshot)
   121	        mask = torch.zeros(snapshot.x.size(0), dtype=torch.bool, device=device)
   122	        mask[active_ids] = True
   123	        
   124	        loss = F.nll_loss(logits[mask], snapshot.y[mask])
   125	        loss.backward()
   126	        optimizer.step()
   127	
   128	        # Monitor results
   129	        with torch.no_grad():
   130	            _, pred = torch.max(logits[mask], dim=1)
   131	            all_preds.extend(pred.cpu().numpy())
   132	            all_true.extend(snapshot.y[mask].cpu().numpy())
   133	
   134	        if (i+1) % 5 == 0:
   135	            print(f"Snapshot {i+1} | Loss: {loss.item():.4f}")
   136	
   137	    print("\n" + "="*50 + "\nOFFICIAL PAPER REPORT\n" + "="*50)
   138	    print(classification_report(all_true, all_preds, digits=4))

     1	import pandas as pd
     2	import torch
     3	import torch.nn.functional as F
     4	import os
     5	import glob
     6	import numpy as np
     7	import matplotlib.pyplot as plt
     8	from torch_geometric.data import Data
     9	from torch_geometric.nn import GATConv
    10	from sklearn.metrics import classification_report
    11	from imblearn.over_sampling import SMOTE
    12	
    13	# ==========================================
    14	# 1. THE COGNITIVE GAT MODEL
    15	# ==========================================
    16	class CognitiveGAT(torch.nn.Module):
    17	    def __init__(self, in_channels, out_channels, history_weight=0.3):
    18	        super(CognitiveGAT, self).__init__()
    19	        self.conv1 = GATConv(in_channels, 16, heads=8, concat=True, dropout=0.2)
    20	        self.conv2 = GATConv(16 * 8, out_channels, heads=1, concat=False)
    21	        self.history_weight = history_weight
    22	        self.previous_trust_scores = None
    23	
    24	    def forward(self, data):
    25	        x, edge_index = data.x, data.edge_index
    26	        x, (edge_ids, alpha) = self.conv1(x, edge_index, return_attention_weights=True)
    27	        x = F.elu(x)
    28	        
    29	        current_trust = alpha.mean(dim=1)
    30	        if self.previous_trust_scores is not None and self.previous_trust_scores.size(0) == current_trust.size(0):
    31	            blended_trust = (1 - self.history_weight) * current_trust + \
    32	                            (self.history_weight) * self.previous_trust_scores
    33	        else:
    34	            blended_trust = current_trust
    35	        
    36	        self.previous_trust_scores = blended_trust.detach()
    37	        x = F.dropout(x, p=0.3, training=self.training)
    38	        x = self.conv2(x, edge_index)
    39	        return F.log_softmax(x, dim=1), blended_trust
    40	
    41	# ==========================================
    42	# 2. DATA LOADER WITH SMOTE & GRAPH MAPPING
    43	# ==========================================
    44	class BoTIoTFullLoader:
    45	    def __init__(self, base_path="/home/rguig/BoT-IoT"):
    46	        self.base_path = base_path
    47	        names_path = os.path.join(self.base_path, "data_names.csv")
    48	        names_df = pd.read_csv(names_path)
    49	        self.headers = names_df.columns.tolist()
    50	        self.ip_map = {}
    51	        self.node_count = 0
    52	        # Reduced sampling strategy to 0.3 to prevent over-synthetic noise
    53	        self.smote = SMOTE(sampling_strategy=0.3, random_state=42)
    54	
    55	    def get_node_id(self, ip):
    56	        ip_str = str(ip)
    57	        if ip_str not in self.ip_map:
    58	            self.ip_map[ip_str] = self.node_count
    59	            self.node_count += 1
    60	        return self.ip_map[ip_str]
    61	
    62	    def stream_snapshots(self):
    63	        all_csvs = sorted(glob.glob(os.path.join(self.base_path, "*.csv")))
    64	        data_files = [f for f in all_csvs if "data_names" not in f]
    65	
    66	        for file in data_files:
    67	            df = pd.read_csv(file, header=None, names=self.headers, low_memory=False)
    68	            df['attack'] = pd.to_numeric(df['attack'], errors='coerce').fillna(0).astype(int)
    69	            cols = ['pkts', 'bytes', 'seq', 'dur']
    70	            df[cols] = df[cols].apply(pd.to_numeric, errors='coerce').fillna(0)
    71	            
    72	            try:
    73	                X_resampled, y_resampled = self.smote.fit_resample(df[cols], df['attack'])
    74	                df_res = pd.DataFrame(X_resampled, columns=cols)
    75	                df_res['attack'] = y_resampled
    76	                
    77	                attack_ips = df[df['attack'] == 1][['saddr', 'daddr']]
    78	                if not attack_ips.empty:
    79	                    needed = len(df_res) - len(df)
    80	                    if needed > 0:
    81	                        extra_ips = attack_ips.sample(n=needed, replace=True).reset_index(drop=True)
    82	                        full_saddr = pd.concat([df['saddr'], extra_ips['saddr']]).reset_index(drop=True)
    83	                        df_res['saddr'] = full_saddr
    84	                    else:
    85	                        df_res['saddr'] = df['saddr']
    86	            except Exception:
    87	                df_res = df
    88	
    89	            # Logic: Connect each flow to its Source IP node
    90	            src = [i for i in range(len(df_res))]
    91	            dst = [self.get_node_id(ip) % len(df_res) for ip in df_res['saddr']]
    92	            edge_index = torch.tensor([src, dst], dtype=torch.long)
    93	
    94	            x = torch.log1p(torch.from_numpy(df_res[cols].values.astype(np.float32)))
    95	            y = torch.tensor(df_res['attack'].values, dtype=torch.long)
    96	            yield Data(x=x, edge_index=edge_index, y=y), os.path.basename(file)
    97	
    98	# ==========================================
    99	# 3. BALANCED TRAINING ENGINE
   100	# ==========================================
   101	if __name__ == "__main__":
   102	    loader = BoTIoTFullLoader()
   103	    model = CognitiveGAT(in_channels=4, out_channels=2)
   104	    
   105	    # ADJUSTED WEIGHTS: Changed from 50.0 to 5.0 to balance Benign vs Attack
   106	    class_weights = torch.tensor([1.0, 2.0]) 
   107	    optimizer = torch.optim.Adam(model.parameters(), lr=0.001)
   108	    
   109	    all_preds, all_true = [], []
   110	    trust_history = []
   111	
   112	    print(" Running Optimized Cog-GAT Pipeline...")
   113	
   114	    for i, (snapshot, filename) in enumerate(loader.stream_snapshots()):
   115	        model.train()
   116	        optimizer.zero_grad()
   117	        out, trust_scores = model(snapshot)
   118	        
   119	        avg_trust = trust_scores.mean().item()
   120	        dpi_savings = (trust_scores > 0.65).float().mean().item() * 100
   121	        
   122	        loss = F.nll_loss(out, snapshot.y, weight=class_weights)
   123	        loss.backward()
   124	        optimizer.step()
   125	
   126	        trust_history.append({'snapshot_index': i, 'avg_trust': avg_trust, 'dpi_savings': dpi_savings})
   127	        _, pred = torch.max(out, dim=1)
   128	        all_preds.extend(pred.cpu().numpy())
   129	        all_true.extend(snapshot.y.cpu().numpy())
   130	
   131	        if (i+1) % 5 == 0:
   132	            print(f"[{i+1}/74] Snap: {filename} | Trust: {avg_trust:.4f} | Savings: {dpi_savings:.1f}%")
   133	
   134	    print("\n" + "="*50 + "\nBALANCED PERFORMANCE METRICS\n" + "="*50)
   135	    print(classification_report(all_true, all_preds, target_names=['Benign', 'Attack']))
   136	    
   137	    df_results = pd.DataFrame(trust_history)
   138	    df_results.to_csv("balanced_cog_gat_results.csv", index=False)
